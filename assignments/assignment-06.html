import pandas as pd
from scipy import stats

df = pd.read_csv("./data/clean.csv")
x = df['feature_alpha']
y = df['primary_metric']

# Pearson r and regression slope
r, p_val = stats.pearsonr(x, y)
slope, intercept, r_val, p_val, std_err = stats.linregress(x, y)

print(f"Regression Line: y = {slope:.3f}x + {intercept:.3f} (r={r:.3f}, R²={r_val**2:.3f})")
