import pandas as pd
from statsmodels.tsa.seasonal import seasonal_decompose

df = pd.read_csv("./data/clean.csv", parse_dates=['timestamp_utc'], index_col='timestamp_utc')
ts = df['primary_metric'].resample('M').mean()

# Classical additive decomposition
decomposition = seasonal_decompose(ts, model='additive', period=12)
print("Decomposition completed: Trend, Seasonal, and Residual components extracted.")
