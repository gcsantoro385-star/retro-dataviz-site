import numpy as np
import pandas as pd

df = pd.read_csv("./data/clean.csv")
data = df['primary_metric']

# Compute Tukey Fences
q25, q75 = np.percentile(data, [25, 75])
iqr = q75 - q25
lower_fence = q25 - (1.5 * iqr)
upper_fence = q75 + (1.5 * iqr)

outliers = df[(data < lower_fence) | (data > upper_fence)]
print(f"[OUTLIERS] Flagged {len(outliers):,} records outside [{lower_fence:.1f}, {upper_fence:.1f}]")
