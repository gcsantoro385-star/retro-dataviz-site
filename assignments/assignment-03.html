import pandas as pd
from scipy import stats

df = pd.read_csv("./data/raw_empirical_records.csv")

# Compute comprehensive statistical profile
metrics = df['primary_metric'].dropna()

mean_val = metrics.mean()
median_val = metrics.median()
std_val = metrics.std()
skew_val = stats.skew(metrics)
kurt_val = stats.kurtosis(metrics)

print(f"Summary Statistics: Mean={mean_val:.2f}, Median={median_val:.2f}, Skew={skew_val:.2f}")
