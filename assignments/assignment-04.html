import pandas as pd
import numpy as np

def clean_empirical_dataset(input_file: str, output_file: str):
    df = pd.read_csv(input_file)
    
    # 1. Clean categoricals
    df['region_code'] = df['region_code'].astype(str).str.strip().str.upper()
    
    # 2. Impute continuous values by regional median
    df['primary_metric'] = df.groupby('region_code')['primary_metric'].transform(
        lambda grp: grp.fillna(grp.median())
    )
    
    # 3. Engineer standardized log transform for heavy-tailed metric
    df['metric_log1p'] = np.log1p(np.maximum(df['primary_metric'], 0))
    
    df.to_csv(output_file, index=False)
    print("[SUCCESS] Cleaned dataset serialized to:", output_file)

clean_empirical_dataset("./data/raw.csv", "./data/clean.csv")
