import pandas as pd

def audit_schema(csv_path: str):
    df = pd.read_csv(csv_path, parse_dates=['timestamp_utc'])
    
    # Missing value evaluation
    null_summary = df.isnull().sum()
    null_pct = (null_summary / len(df)) * 100
    
    # Deduplication integrity
    dupes = df.duplicated(subset=['record_id']).sum()
    print(f"[AUDIT] Duplicate record_id count: {dupes}")
    
    return pd.DataFrame({'null_count': null_summary, 'null_percent': null_pct})

report = audit_schema("./data/raw_empirical_records.csv")
print(report[report['null_count'] > 0])
